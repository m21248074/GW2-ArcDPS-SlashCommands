cmake_minimum_required(VERSION 3.24)
project(arcdps_slashcommands VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Requires Windows x64 and MSVC")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(imgui STATIC third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp
  third_party/imgui/imgui_tables.cpp third_party/imgui/imgui_widgets.cpp)
target_include_directories(imgui PUBLIC third_party/imgui)
add_library(command_locator STATIC src/locator.cpp)
target_include_directories(command_locator PUBLIC src)
add_library(arcdps_slashcommands SHARED src/plugin.cpp src/engine_tick.cpp)
add_library(minhook STATIC third_party/minhook/src/buffer.c third_party/minhook/src/hook.c
  third_party/minhook/src/trampoline.c third_party/minhook/src/hde/hde64.c)
target_include_directories(minhook PUBLIC third_party/minhook/include)
target_compile_definitions(minhook PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(arcdps_slashcommands PRIVATE command_locator imgui minhook shell32)
add_executable(locator_tests tests/locator_tests.cpp)
target_link_libraries(locator_tests PRIVATE command_locator)
add_executable(dll_contract tests/dll_contract.cpp)
target_link_libraries(dll_contract PRIVATE imgui)
add_dependencies(dll_contract arcdps_slashcommands)
add_executable(command_tests tests/command_tests.cpp src/engine_tick.cpp)
target_link_libraries(command_tests PRIVATE command_locator imgui minhook shell32)
add_executable(engine_tick_tests tests/engine_tick_tests.cpp src/engine_tick.cpp)
target_link_libraries(engine_tick_tests PRIVATE minhook)
add_executable(locator_probe tests/locator_probe.cpp)
target_link_libraries(locator_probe PRIVATE command_locator)
foreach(t imgui command_locator arcdps_slashcommands locator_tests dll_contract command_tests engine_tick_tests locator_probe)
  target_compile_definitions(${t} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${t} PRIVATE /utf-8 /W4 /EHsc)
endforeach()
enable_testing()
add_test(NAME signature_validation COMMAND locator_tests)
add_test(NAME dll_imgui_contract COMMAND dll_contract $<TARGET_FILE:arcdps_slashcommands>)
add_test(NAME command_queue_settings COMMAND command_tests)
add_test(NAME engine_tick_dispatch_abi_lifecycle COMMAND engine_tick_tests)
